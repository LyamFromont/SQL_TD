Création TRIGGER trg_verif_salaire

Interdit la diminution du salaire d'un employé.

create or replace TRIGGER trg_verif_salaire
AFTER UPDATE OF salaire ON EMPLOYE
FOR EACH ROW
BEGIN
if :new.salaire < :old.salaire then
raise_application_error(-20001,'viol de la règle :un salaire ne peut etre diminué') ;
end if ;
END;

Création TRIGGER_SUM_HEBDO & TRIGGER_SUM_HEBDO2

La somme des durées ne doit pas excéder le temps hebdo.

create or replace TRIGGER TRIGGER_SUM_HEBDO AFTER INSERT OR UPDATE OF DUREE ON TRAVAIL
DECLARE
REC_employe employe%ROWTYPE;
BEGIN
Select * INTO REC_employe from employe e
where (select sum(duree) from travail t where e.nuempl=t.nuempl)> hebdo ; 
raise_application_error(-20001,'viol de la règle 1 erreur :trop d heure par rapport a la limite hebdo') ;
EXCEPTION
WHEN NO_DATA_FOUND THEN NULL;
WHEN TOO_MANY_ROWS THEN raise_application_error(-20001,'viol de la règle plusieurs erreurs:trop d heure par rapport a la limite hebdo') ;
END;

Création TRIGGER_RESP_MAX

Un employé est responsable au plus sur 3 projets.

create or replace TRIGGER "TRIGGER_RESP_MAX" AFTER INSERT OR UPDATE OF RESP ON PROJET

REC_employe employe%ROWTYPE;
BEGIN
Select * INTO REC_employe from employe e

WHEN NO_DATA_FOUND THEN NULL;

Création TRIGGER_2PROJET

Un service ne peut être concerné par plus de 3 projets.

create or replace TRIGGER TRIGGER_2PROJET AFTER INSERT OR UPDATE OF NUSERV ON CONCERNE 
DECLARE 
REC_service service%ROWTYPE; 
BEGIN
SELECT * INTO REC_service FROM service s 
WHERE (SELECT COUNT(*) FROM concerne c WHERE c.nuserv = s.nuserv) > 2; 
raise_application_error(-20001,'viol de la règle 1 erreur:trop de projet pour un service') ;
EXCEPTION
WHEN NO_DATA_FOUND THEN NULL;
WHEN TOO_MANY_ROWS THEN raise_application_error(-20001,'viol de la règle plusieurs erreurs:trop de projet pour un service') ;
END;

Création TRIGGER_SALAIRE_SERV

Vérifie qu'un employé ne gagne pas plus (ou autant) que son chef de service.

create or replace TRIGGER "TRIGGER_SALAIRE_SERV" AFTER INSERT OR UPDATE OF CHEF ON SERVICE 
DECLARE
REC_service service%ROWTYPE;
BEGIN
SELECT s.* INTO REC_service FROM service s join employe e on e.affect = s.nuserv join employe chef on s.chef = chef.nuempl
where e.salaire > chef.salaire; 
raise_application_error(-20001,'viol de la règle 1 erreur :un employe gagne plus (ou autant ) que son chef') ;
EXCEPTION
WHEN NO_DATA_FOUND THEN NULL;
WHEN TOO_MANY_ROWS THEN raise_application_error(-20001,'viol de la règle plusieur erreur :plusieur employe gagne plus (ou autant ) que son chef') ;
END;

Création TRIGGER_SALAIRE_PROJ

Vérifie qu'un employé ne gagne pas plus (ou autant) que son responsable de projet.

create or replace TRIGGER "TRIGGER_SALAIRE_PROJ" AFTER INSERT OR UPDATE OF resp ON Projet 
DECLARE
REC_projet projet%ROWTYPE;
BEGIN
SELECT p.* INTO REC_projet FROM projet p join travail t on t.nuproj = p.nuproj
join employe e on e.nuempl = t.nuempl
join employe resp on p.resp = resp.nuempl
where e.salaire > resp.salaire;
raise_application_error(-20001,'viol de la règle 1 erreur :un employe gagne plus (ou autant ) que son responsable') ;

WHEN NO_DATA_FOUND THEN NULL;
WHEN TOO_MANY_ROWS THEN raise_application_error(-20001,'viol de la règle plusieur erreur :plusieur employe gagne plus (ou autant ) que son responsable') ;
END;
